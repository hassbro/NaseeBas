        function playSound(freq, duration = 0.08, type = 'sine') {

            try {

                if (!audioCtx) audioCtx = new AudioCtx();

                if (audioCtx.state === 'suspended') audioCtx.resume();


                const osc = audioCtx.createOscillator();

                const gain = audioCtx.createGain();

                osc.type = type;

                osc.frequency.setValueAtTime(freq, audioCtx.currentTime);

                gain.gain.setValueAtTime(0.12, audioCtx.currentTime);

                gain.gain.exponentialRampToValueAtTime(0.001, audioCtx.currentTime + duration);

                osc.connect(gain);

                gain.connect(audioCtx.destination);

                osc.start();

                osc.stop(audioCtx.currentTime + duration);

            } catch (e) {

                // Audio context restriction silently handled

            }

        }

        function handleLetterInput(char) {

            if (isGameOver) return;


            if (currentTileIndex < 0) {

                currentTileIndex = 0;

            }


            playSound(320, 0.05, 'triangle');

            boardState[currentRow][currentTileIndex].base = char;

            boardState[currentRow][currentTileIndex].fili = ''; // Reset fili when selecting a new base letter

            updateTileDisplay(currentRow, currentTileIndex);

        }


        function handleFiliInput(fili) {

            if (isGameOver) return;

            if (currentTileIndex < 0) return;


            const currentCell = boardState[currentRow][currentTileIndex];


            if (currentCell.base) {

                playSound(440, 0.06, 'sine');

                currentCell.fili = fili;

                updateTileDisplay(currentRow, currentTileIndex);


                // Move right-to-left (towards index 0) automatically when fili is added

                if (currentTileIndex > 0) {

                    setActiveTile(currentTileIndex - 1);

                }

            } else {

                playSound(200, 0.1, 'sawtooth');

                showToast("ފުރަތަމަ އަކުރެއް ނަންގަވާ!");

            }

        }


        function handleBackspace() {

            if (isGameOver) return;


            playSound(260, 0.05, 'square');

            const currentCell = boardState[currentRow][currentTileIndex];


            if (currentCell.fili) {

                currentCell.fili = '';

                updateTileDisplay(currentRow, currentTileIndex);

            } 

            else if (currentCell.base) {

                currentCell.base = '';

                updateTileDisplay(currentRow, currentTileIndex);

            } 

            else if (currentTileIndex < wordLength - 1) {

                setActiveTile(currentTileIndex + 1);

                const prevCell = boardState[currentRow][currentTileIndex];

                if (prevCell.fili) {

                    prevCell.fili = '';

                } else {

                    prevCell.base = '';

                }

                updateTileDisplay(currentRow, currentTileIndex);

            }

        }


        function updateTileDisplay(row, col) {

            const tileEl = document.getElementById(`tile-${row}-${col}`);

            const textSpan = document.getElementById(`tile-text-${row}-${col}`);

            const cell = boardState[row][col];

            

            const combinedText = (cell.base || '') + (cell.fili || '');

            textSpan.textContent = combinedText;


            if (combinedText) {

                tileEl.classList.add('filled');

            } else {

                tileEl.classList.remove('filled');

            }

        }


        function handleEnter() {

            if (isGameOver) return;


            const currentRowData = boardState[currentRow];

            const isRowComplete = currentRowData.every(tile => tile.base !== '' && tile.fili !== '');


            if (!isRowComplete) {

                playSound(180, 0.15, 'sawtooth');

                showToast("ކޮންމެ އަކުރަކަށް ފިއްޔެއް ޖައްސަވާ!");

                return;

            }


            evaluateRowGuesses();

        }


        function evaluateRowGuesses() {

            const guessRow = boardState[currentRow];

            

            const letterCounts = {};

            const filiCounts = {};


            targetWordObj.forEach(t => {

                letterCounts[t.base] = (letterCounts[t.base] || 0) + 1;

                filiCounts[t.fili] = (filiCounts[t.fili] || 0) + 1;

            });


            const letterResults = Array(wordLength).fill('absent');

            const filiResults = Array(wordLength).fill('absent');


            // --- PASS 1: Identify EXACT Matches (Green) ---

            guessRow.forEach((tile, i) => {

                if (tile.base === targetWordObj[i].base) {

                    letterResults[i] = 'correct';

                    letterCounts[tile.base]--;

                }

                if (tile.fili === targetWordObj[i].fili) {

                    filiResults[i] = 'correct';

                    filiCounts[tile.fili]--;

                }

            });


            // --- PASS 2: Identify PRESENT Matches (Yellow) ---

            guessRow.forEach((tile, i) => {

                if (letterResults[i] !== 'correct') {

                    if (letterCounts[tile.base] && letterCounts[tile.base] > 0) {

                        letterResults[i] = 'present';

                        letterCounts[tile.base]--;

                    }

                }

                if (filiResults[i] !== 'correct') {

                    if (filiCounts[tile.fili] && filiCounts[tile.fili] > 0) {

                        filiResults[i] = 'present';

                        filiCounts[tile.fili]--;

                    }

                }

            });


            // --- Apply Visual Updates with Reveal Flip Animation ---

            guessRow.forEach((tile, colIndex) => {

                const tileEl = document.getElementById(`tile-${currentRow}-${colIndex}`);

                const zoneFili = document.getElementById(`zone-fili-${currentRow}-${colIndex}`);

                const zoneLetter = document.getElementById(`zone-letter-${currentRow}-${colIndex}`);


                const lRes = letterResults[colIndex];

                const fRes = filiResults[colIndex];


                setTimeout(() => {

                    playSound(400 + colIndex * 60, 0.08, 'sine');

                    tileEl.classList.add('flip');


                    if (lRes === 'correct' && fRes === 'correct') {

                        zoneFili.className = 'zone-fili bg-correct';

                        zoneLetter.className = 'zone-letter bg-correct';

                        tileEl.className = 'tile thaana-text border-correct';

                    } else {

                        zoneFili.className = `zone-fili bg-${fRes}`;

                        zoneLetter.className = `zone-letter bg-${lRes}`;


                        if (lRes === 'correct' || fRes === 'correct') {

                            tileEl.style.borderColor = '#538d4e';

                        } else if (lRes === 'present' || fRes === 'present') {

                            tileEl.style.borderColor = '#b59f3b';

                        } else {

                            tileEl.style.borderColor = '#3a3a3c';

                        }

                    }


                    markKeyStatus(tile.base, lRes);

                    markFiliStatus(tile.fili, fRes);


                }, colIndex * 180);

            });


            const isWin = letterResults.every(res => res === 'correct') && filiResults.every(res => res === 'correct');


            setTimeout(() => {

                stats.played++;

                if (isWin) {

                    isGameOver = true;

                    stats.wins++;

                    stats.streak++;

                    if (stats.streak > stats.maxStreak) stats.maxStreak = stats.streak;

                    saveStats();

                    playSound(600, 0.2, 'sine');

                    setTimeout(() => playSound(800, 0.3, 'sine'), 150);

                    showEndGameModal(true);

                } else if (currentRow === maxRows - 1) {

                    isGameOver = true;

                    stats.streak = 0;

                    saveStats();

                    playSound(200, 0.3, 'sawtooth');

                    showEndGameModal(false);

                } else {

                    currentRow++;

                    currentTileIndex = wordLength - 1; // Reset to rightmost tile for next row

                    setActiveTile(wordLength - 1);

                }

                updateStatsModal();

            }, wordLength * 180 + 100);

        }

        function parseThaanaWordString(text) {

            if (!text) return [];


            let cleaned = text.replace(/[^\u0780-\u07BF]/g, '').trim();

            if (!cleaned) return [];


            let chars = Array.from(cleaned);


            let filiFirstCount = 0;

            let letterFirstCount = 0;


            for (let i = 0; i < chars.length - 1; i++) {

                if (FILI_LIST.includes(chars[i]) && THAANA_LETTERS.includes(chars[i + 1])) filiFirstCount++;

                if (THAANA_LETTERS.includes(chars[i]) && FILI_LIST.includes(chars[i + 1])) letterFirstCount++;

            }


            if (filiFirstCount > letterFirstCount) {

                chars.reverse();

            }


            const result = [];

            let i = 0;

            while (i < chars.length) {

                const char = chars[i];

                if (THAANA_LETTERS.includes(char)) {

                    let base = char;

                    let fili = '';

                    if (i + 1 < chars.length && FILI_LIST.includes(chars[i + 1])) {

                        fili = chars[i + 1];

                        i += 2;

                    } else {

                        fili = (i === chars.length - 1) ? 'ް' : 'ަ';

                        i += 1;

                    }

                    result.push({ base, fili });

                } else {

                    i++;

                }

            }

            return result;

        }


        function isValidRadheefHeadword(wordText, parsed) {

            if (!wordText || parsed.length < 3 || parsed.length > 5) return false;

            return /^[\u0780-\u07BF]+$/.test(wordText);

        }


        const DHIVEHI_POS_TAGS = ['ނ.', 'ނއ.', 'ކ.', 'ކއ.', 'އނ.', 'ނއަ.', 'މނ.', 'ނނ.', 'ބ.', 'އކ.'];


        function getDhivehiPosTag(token) {

            if (!token) return null;

            const clean = token.trim();

            if (DHIVEHI_POS_TAGS.includes(clean)) return clean;

            const cleanWithDot = clean.endsWith('.') ? clean : clean + '.';

            if (DHIVEHI_POS_TAGS.includes(cleanWithDot)) return cleanWithDot;

            return null;

        }


        function isOriginTag(token) {

            if (!token) return false;

            const clean = token.trim();

            return getDhivehiPosTag(clean) !== null || (clean.length === 1 && /^[\u0780-\u07B1]$/.test(clean));

        }


        function parseAndAddRawDictionaryText(text, statusEl, sourceName) {

            const lines = text.split(/\r?\n/);

            const newDicts = { 3: [], 4: [], 5: [] };

            const seenWords = new Set();

            const MAX_WORDS_PER_MODE = 400;


            for (let line of lines) {

                line = line.trim();

                if (!line) continue;


                const tokens = line.split(/\s+/);

                if (tokens.length >= 2) {

                    const firstToken = tokens[0].replace(/[.,،؛()!"'\\/]/g, '').trim();

                    const parsed = parseThaanaWordString(firstToken);


                    if (parsed.length >= 3 && parsed.length <= 5) {

                        const len = parsed.length;

                        const reconstructedText = parsed.map(p => p.base + p.fili).join('');


                        if (newDicts[len] && newDicts[len].length < MAX_WORDS_PER_MODE && !seenWords.has(reconstructedText)) {

                            let posTag = "";

                            let meaningStartIndex = 1;


                            const detectedPos = getDhivehiPosTag(tokens[1]);

                            if (detectedPos) {

                                posTag = detectedPos;

                                meaningStartIndex = 2;

                            }


                            const meaningText = tokens.slice(meaningStartIndex).join(' ').trim() || `${reconstructedText} - ${sourceName} ނެގިފައިވާ ލަފްޒެއް`;


                            newDicts[len].push({

                                word: parsed,

                                text: reconstructedText,

                                pos: posTag,

                                meaning: meaningText

                            });

                            seenWords.add(reconstructedText);

                        }

                    }

                }

            }


            let totalAdded = 0;

            [3, 4, 5].forEach(mode => {

                if (newDicts[mode] && newDicts[mode].length > 0) {

                    WORD_DICTIONARIES[mode] = newDicts[mode];

                    totalAdded += newDicts[mode].length;

                }

            });


            if (totalAdded > 0) {

                try {

                    localStorage.setItem('thaana_pdf_dictionaries', JSON.stringify(WORD_DICTIONARIES));

                } catch (e) {

                    console.warn("LocalStorage quota full", e);

                }

                statusEl.textContent = `${totalAdded} ލަފްޒު އެޅިއްޖެ!`;

                showToast(`${sourceName} ލަފްޒުތައް އިތުރުކުރެވިއްޖެ!`);

                initRadheefModal();

                initGameMode(currentMode);

            } else {

                statusEl.textContent = "3، 4 ނުވަތަ 5 އަކުރުގެ ލަފްޒެއް ނުފެނުނު";

                showToast("އެކަށީގެންވާ ދިގުމިނުގެ ލަފްޒެއް ނުފެނުނު");

                const arrayBuffer = await file.arrayBuffer();

                const loadingTask = pdfjsLib.getDocument({ data: arrayBuffer });

                const pdf = await loadingTask.promise;

                

                const startPage = pdf.numPages >= 12 ? 12 : 1;

                const endPage = Math.min(pdf.numPages, 1190);

                

                const newDicts = { 3: [], 4: [], 5: [] };

                const seenWords = new Set();

                const MAX_WORDS_PER_MODE = 250;

                    let pageTextItems = [];

                    try {

                        const page = await pdf.getPage(pageNum);

                        const textContent = await page.getTextContent();

                        if (textContent && textContent.items) {

                            pageTextItems = textContent.items.map(item => item.str ? item.str.trim() : '').filter(Boolean);

                        }

                    }

                        statusEl.textContent = `PDF ކިޔެނީ (${pageNum} / ${endPage})...`;

                        await new Promise(r => setTimeout(r, 0));

                    }


                    if (newDicts[3].length >= MAX_WORDS_PER_MODE &&

                        newDicts[4].length >= MAX_WORDS_PER_MODE &&

                        newDicts[5].length >= MAX_WORDS_PER_MODE) {

                    }

                    const meaningsText = document.getElementById('bulk-meanings-input')?.value || '';

                    const posText = document.getElementById('bulk-pos-input')?.value || '';


                    const wordsList = wordsText.split(/\r?\n/).map(s => s.trim()).filter(Boolean);

                    const meaningsList = meaningsText.split(/\r?\n/).map(s => s.trim()).filter(Boolean);

                        return;

                    }


                    wordsList.forEach((rawWord, idx) => {

                        const parsed = parseThaanaWordString(rawWord);

                        const wordLen = parsed.length;


                        if (wordLen >= 3 && wordLen <= 5) {

                            const reconstructedText = parsed.map(p => p.base + p.fili).join('');

                            const meaning = meaningsList[idx] || meaningsList[0] || `${reconstructedText} ގެ މާނަ`;

                            const pos = posList[idx] || posList[0] || '';


                            const exists = WORD_DICTIONARIES[wordLen].some(w => w.text === reconstructedText);

                            if (!exists) {

                                WORD_DICTIONARIES[wordLen].push({

                                    word: parsed,

                                    text: reconstructedText,

                                    pos: pos,

                                    meaning: meaning

                                });

                                importedCount[wordLen]++;

                                totalAdded++;

                            }

                    });

                            const parsed = parseThaanaWordString(rawWord);

                            const wordLen = parsed.length;


                            if (wordLen >= 3 && wordLen <= 5) {

                                const reconstructedText = parsed.map(p => p.base + p.fili).join('');

                                let pos = "";

                                let meaning = "";

                                    meaning = parts[1];

                                } else if (parts.length >= 3) {

                                    pos = parts[1];

                                    meaning = parts.slice(2).join(' ');

                                } else {

                                    meaning = `${reconstructedText} ގެ މާނަ`;

                                }


                                const exists = WORD_DICTIONARIES[wordLen].some(w => w.text === reconstructedText);

                                        meaning: meaning

                                    totalAdded++;

                                }
